1 Write a program in C to input a string and print it.

Question

Write a program in C to input a string and print it.

Solution

#include <stdio.h>

int main() {
    char str[100];

    printf("Input the string: ");
    fgets(str, sizeof(str), stdin);

    printf("The string you entered is: %s", str);

    return 0;
}

Explanation

  • char str[100] creates a character array that can store a string.
  • fgets() is used so that the input can contain spaces.
  • The entered string is then printed using %s.
2 Perform string copy with and without library functions.

Question

Perform string copy with and without library functions.

Solution 1 — Using the Library Function

The standard library function strcpy() can be used to copy one string into another.

#include <stdio.h>
#include <string.h>

int main() {
    char source[100];
    char destination[100];

    printf("Enter the source string: ");
    fgets(source, sizeof(source), stdin);

    strcpy(destination, source);

    printf("Copied string: %s", destination);

    return 0;
}

Solution 2 — Without Using a Library Function

Here, the characters are copied one by one.

#include <stdio.h>

int main() {
    char source[100];
    char destination[100];
    int i = 0;

    printf("Enter the source string: ");
    fgets(source, sizeof(source), stdin);

    while (source[i] != '\0') {
        destination[i] = source[i];
        i++;
    }

    destination[i] = '\0';

    printf("Copied string: %s", destination);

    return 0;
}

Explanation

A C string ends with the null character '\0'. Therefore, while copying manually, we copy characters until '\0' is reached and then add '\0' to the destination string.

3 Write a program in C to find the length of a string with and without using library functions.

Question

Write a program in C to find the length of a string with and without using library functions.

Solution 1 — Using strlen()

#include <stdio.h>
#include <string.h>

int main() {
    char str[100];

    printf("Input the string: ");
    fgets(str, sizeof(str), stdin);

    /* Remove the newline character */
    str[strcspn(str, "\n")] = '\0';

    printf("Length of the string is: %zu\n", strlen(str));

    return 0;
}

Solution 2 — Without Using strlen()

#include <stdio.h>

int main() {
    char str[100];
    int length = 0;

    printf("Input the string: ");
    fgets(str, sizeof(str), stdin);

    while (str[length] != '\0' && str[length] != '\n') {
        length++;
    }

    printf("Length of the string is: %d\n", length);

    return 0;
}

Explanation

A string's length is the number of characters before the null character '\0'. The newline inserted by fgets() is excluded in the second solution.

4 Write a program in C to compare two strings without using string library functions.

Question

Write a program in C to compare two strings without using string library functions.

Solution

#include <stdio.h>

int main() {
    char str1[100], str2[100];
    int i = 0;
    int equal = 1;

    printf("Input the 1st string: ");
    fgets(str1, sizeof(str1), stdin);

    printf("Input the 2nd string: ");
    fgets(str2, sizeof(str2), stdin);

    while (str1[i] != '\0' || str2[i] != '\0') {
        if (str1[i] == '\n') {
            if (str2[i] == '\n' || str2[i] == '\0')
                break;
        }

        if (str1[i] != str2[i]) {
            equal = 0;
            break;
        }

        i++;
    }

    if (equal)
        printf("Strings are equal.\n");
    else
        printf("Strings are not equal.\n");

    return 0;
}

Explanation

The program compares corresponding characters of the two strings one by one. If any pair of characters is different, the strings are not equal.

5 Write a program in C to print only the alphabets from the string and then the length.

Question

Write a program in C to print only the alphabets from the string and then the length.

Input:

Hello, this is 'algorithm lab'

Expected Output:

Hello this is algorithm lab
23

Solution

#include <stdio.h>
#include <ctype.h>

int main() {
    char str[200];
    int i;
    int length = 0;

    printf("Input: ");
    fgets(str, sizeof(str), stdin);

    printf("Output: ");

    for (i = 0; str[i] != '\0'; i++) {
        if (isalpha((unsigned char)str[i]) || str[i] == ' ') {
            printf("%c", str[i]);

            if (isalpha((unsigned char)str[i])) {
                length++;
            }
        }
    }

    printf("\n%d\n", length);

    return 0;
}

Explanation

  • isalpha() checks whether a character is an alphabet.
  • Spaces are also printed so that the words remain separated.
  • Only alphabetic characters are counted in the length.
  • For the given input, the number of alphabetic characters is 23.
6 Write a program in C to count the total number of words in a string.

Question

Write a program in C to count the total number of words in a string.

Solution

#include <stdio.h>
#include <ctype.h>

int main() {
    char str[200];
    int i;
    int words = 0;
    int inWord = 0;

    printf("Input the string: ");
    fgets(str, sizeof(str), stdin);

    for (i = 0; str[i] != '\0'; i++) {
        if (!isspace((unsigned char)str[i]) && inWord == 0) {
            words++;
            inWord = 1;
        }
        else if (isspace((unsigned char)str[i])) {
            inWord = 0;
        }
    }

    printf("Total number of words in the string is: %d\n", words);

    return 0;
}

Explanation

A new word begins when: 1. The current character is not whitespace. 2. The previous position was not already inside a word.

This approach also handles multiple spaces correctly.

7 Create a function in C to count the total number of alphabets, digits, and special characters in a string.

Question

Create a function in C to count the total number of alphabets, digits, and special characters in a string.

Solution

#include <stdio.h>
#include <ctype.h>

void countCharacters(char str[]) {
    int alphabets = 0;
    int digits = 0;
    int special = 0;
    int i;

    for (i = 0; str[i] != '\0' && str[i] != '\n'; i++) {
        if (isalpha((unsigned char)str[i])) {
            alphabets++;
        }
        else if (isdigit((unsigned char)str[i])) {
            digits++;
        }
        else {
            special++;
        }
    }

    printf("Number of Alphabets in the string is: %d\n", alphabets);
    printf("Number of Digits in the string is: %d\n", digits);
    printf("Number of Special characters in the string is: %d\n", special);
}

int main() {
    char str[200];

    printf("Input the string: ");
    fgets(str, sizeof(str), stdin);

    countCharacters(str);

    return 0;
}

8 Write a program in C to count the total number of vowels or consonants in a string.

Question

Write a program in C to count the total number of vowels or consonants in a string.

Solution

#include <stdio.h>
#include <ctype.h>

int main() {
    char str[200];
    int vowels = 0;
    int consonants = 0;
    int i;

    printf("Input the string: ");
    fgets(str, sizeof(str), stdin);

    for (i = 0; str[i] != '\0'; i++) {
        char ch = tolower((unsigned char)str[i]);

        if (ch >= 'a' && ch <= 'z') {
            if (ch == 'a' || ch == 'e' || ch == 'i' ||
                ch == 'o' || ch == 'u') {
                vowels++;
            }
            else {
                consonants++;
            }
        }
    }

    printf("The total number of vowels in the string is: %d\n", vowels);
    printf("The total number of consonants in the string is: %d\n", consonants);

    return 0;
}

Explanation

Only alphabetic characters are considered. Therefore, spaces, digits, and punctuation are ignored.

For the given input: - Vowels = 9 - Consonants = 12

9 Write a program in C to find the maximum number of times a character appears in a string.

Question

Write a program in C to find the maximum number of times a character appears in a string.

Solution

#include <stdio.h>

int main() {
    char str[200];
    int frequency[256] = {0};
    int i;
    int max = 0;
    unsigned char maxChar = '\0';

    printf("Input the string: ");
    fgets(str, sizeof(str), stdin);

    for (i = 0; str[i] != '\0'; i++) {
        unsigned char ch = (unsigned char)str[i];

        if (ch != '\n') {
            frequency[ch]++;
        }
    }

    for (i = 0; i < 256; i++) {
        if (frequency[i] > max) {
            max = frequency[i];
            maxChar = (unsigned char)i;
        }
    }

    printf("The Highest frequency of character '%c'\n", maxChar);
    printf("appears number of times: %d\n", max);

    return 0;
}

Explanation

The frequency array stores how many times each character occurs.

For example:

text frequency['e']stores the number of occurrences of e.

The program then searches for the largest frequency.

10 Write a program in C to calculate the sum of numbers from 1 to n using recursion.

Question

Write a program in C to calculate the sum of numbers from 1 to n using recursion.

Solution

#include <stdio.h>

int sum(int n) {
    if (n == 0) {
        return 0;
    }

    return n + sum(n - 1);
}

int main() {
    int n;

    printf("Input the last number of the range starting from 1: ");
    scanf("%d", &n);

    printf("The sum of numbers from 1 to %d: %d\n", n, sum(n));

    return 0;
}

How Recursion Works

For n = 5:

sum(5)
= 5 + sum(4)
= 5 + 4 + sum(3)
= 5 + 4 + 3 + sum(2)
= 5 + 4 + 3 + 2 + sum(1)
= 5 + 4 + 3 + 2 + 1 + sum(0)
= 15

sum(0) is the base case that stops the recursion.

11 Create a recursive function for creating Fibonacci series.

Question

Create a recursive function for creating Fibonacci series.

Solution

#include <stdio.h>

int fibonacci(int n) {
    if (n == 0) {
        return 0;
    }

    if (n == 1) {
        return 1;
    }

    return fibonacci(n - 1) + fibonacci(n - 2);
}

int main() {
    int n, i;

    printf("Enter the number of terms: ");
    scanf("%d", &n);

    printf("Fibonacci Series: ");

    for (i = 0; i < n; i++) {
        printf("%d ", fibonacci(i));
    }

    printf("\n");

    return 0;
}

Explanation

The Fibonacci sequence follows:

F(n) = F(n - 1) + F(n - 2)

```text
F(0) = 0
F(1) = 1
```So the sequence begins:

13 Write a program in C to count the digits of a given number using recursion.

Question

Write a program in C to count the digits of a given number using recursion.

Solution

#include <stdio.h>

int countDigits(long long n) {
    if (n >= -9 && n <= 9) {
        return 1;
    }

    return 1 + countDigits(n / 10);
}

int main() {
    long long n;

    printf("Input a number: ");
    scanf("%lld", &n);

    printf("The number of digits in the number is: %d\n",
           countDigits(n));

    return 0;
}

Explanation

Integer division by 10 removes the last digit.

For example:

text 50 / 10 = 5 5 / 10 = 0Therefore, there are two digits.

The function stops when the remaining number has only one digit.

14 Write a program in C to find the GCD of two numbers using recursion.

Question

Write a program in C to find the GCD of two numbers using recursion.

Solution

#include <stdio.h>

int gcd(int a, int b) {
    if (b == 0) {
        return a;
    }

    return gcd(b, a % b);
}

int main() {
    int a, b;

    printf("Input 1st number: ");
    scanf("%d", &a);

    printf("Input 2nd number: ");
    scanf("%d", &b);

    printf("The GCD of %d and %d is: %d\n", a, b, gcd(a, b));

    return 0;
}

How It Works

The program uses the Euclidean algorithm:

text gcd(a, b) = gcd(b, a % b)For 10 and 50:

text gcd(10, 50) = gcd(50, 10) = gcd(10, 0) = 10So the GCD is 10.

15 Write a program in C to find the Factorial of a number using recursion.

Question

Write a program in C to find the Factorial of a number using recursion.

Solution

#include <stdio.h>

long long factorial(int n) {
    if (n == 0 || n == 1) {
        return 1;
    }

    return n * factorial(n - 1);
}

int main() {
    int n;

    printf("Input a number: ");
    scanf("%d", &n);

    if (n < 0) {
        printf("Factorial is not defined for negative numbers.\n");
    }
    else {
        printf("The Factorial of %d is: %lld\n", n, factorial(n));
    }

    return 0;
}

How Recursion Works

For 5!:

5!
= 5 × 4!
= 5 × 4 × 3!
= 5 × 4 × 3 × 2!
= 5 × 4 × 3 × 2 × 1!
= 120
